   112→          raan: s.elements.raan,

   115→          bstar: s.elements.bstar,

   196→// Screening run: for each protected satellite, screen against all other
   197→// catalog objects within the screening horizon.
   198→export async function runScreening(opts: {
   199→  primaryIds?: string[];
   200→  screeningHorizonDays?: number;
   201→  thresholdKm?: number;
   202→  weights?: RiskWeights;
   203→}): Promise<{ totalEvents: number; criticalCount: number; highCount: number; moderateCount: number; lowCount: number; nextCritical?: ConjunctionDTO; highestRisk?: ConjunctionDTO; conjunctions: ConjunctionDTO[] }> {
   204→  await ensureDemoSeeded();
   205→  const horizonDays = opts.screeningHorizonDays ?? 7;
   206→  const threshold = opts.thresholdKm ?? 5;
   207→  const weights = opts.weights ?? DEFAULT_RISK_WEIGHTS;
   208→  const now = new Date();
   209→  const start = now;
   210→  const end = new Date(now.getTime() + horizonDays * 24 * 3600 * 1000);
   211→
   212→  const primaries = opts.primaryIds
   213→    ? await db.satellite.findMany({ where: { id: { in: opts.primaryIds } } })
   214→    : await db.satellite.findMany({ where: { isProtected: true } });
   215→  const secondaries = await db.satellite.findMany({}); // all catalog objects
   216→
   217→  const allConjunctions: ConjunctionDTO[] = [];
   218→
   219→  for (const prim of primaries) {
   220→    const primEls = elementsFromSat(prim);
   221→    const results: ConjunctionResult[] = [];
   222→    for (const sec of secondaries) {
   223→      if (sec.id === prim.id) continue;
   224→      const secEls = elementsFromSat(sec);
   225→      const conj = screen(primEls, secEls, start, end, threshold);
   226→      if (conj) {
   227→        results.push(conj);
   228→        const risk = computeRisk(conj, prim.epoch, false, sec.objectType, weights);
   229→        const conf = computeConfidence({
   230→          dataAgeHours: (now.getTime() - prim.epoch.getTime()) / 3600000,
   231→          source: prim.source,
   232→          format: prim.format as 'OMM' | 'TLE',
   233→          covarianceAvailable: false,
   234→          propagationHorizonHours: (conj.tca.getTime() - now.getTime()) / 3600000,
   235→          missingMetadata: false,
   236→        });
   237→        // Build timeline
   238→        const timeline = [
   239→          { time: conj.tca.toISOString(), event: 'TCA', detail: `Min range ${conj.minRange.toFixed(2)} km` },
   240→          { time: now.toISOString(), event: 'Event detected', detail: `Risk ${risk.score}/100 (${risk.level})` },
   241→        ];
   242→        const existing = await db.conjunction.findFirst({
   243→          where: { primarySatId: prim.id, secondarySatId: sec.id },
   244→          orderBy: { createdAt: 'desc' },
   245→        });
   246→        if (existing) {
   247→          await db.conjunction.update({
   248→            where: { id: existing.id },
   249→            data: {
   250→              tca: conj.tca,
   251→              minRange: conj.minRange,
   252→              relVelocity: conj.relVelocity,
   253→              screeningStart: start,
   254→              screeningEnd: end,
   255→              screeningThreshold: threshold,
   256→              riskScore: risk.score,
   257→              riskLevel: risk.level,
   258→              confidenceScore: conf.score,
   259→              riskFactorsJson: JSON.stringify(risk.factors),
   260→              timelineJson: JSON.stringify(timeline),
   261→              analysisId: uuidv4(),
   262→            },
   263→          });
   264→        } else {
   265→          await db.conjunction.create({
   266→            data: {
   267→              primarySatId: prim.id,
   268→              secondarySatId: sec.id,
   269→              tca: conj.tca,
   270→              minRange: conj.minRange,
   271→              relVelocity: conj.relVelocity,
   272→              screeningStart: start,
   273→              screeningEnd: end,
   274→              screeningThreshold: threshold,
   275→              riskScore: risk.score,
   276→              riskLevel: risk.level,
   277→              confidenceScore: conf.score,
   278→              riskFactorsJson: JSON.stringify(risk.factors),
   279→              timelineJson: JSON.stringify(timeline),
   280→              analysisId: uuidv4(),
   281→            },
   282→          });
   283→        }
   284→      }
   285→    }
   286→    screeningCache[prim.id] = { lastRun: now, results };
   287→  }
   288→
   289→  // Now read all conjunctions from DB and build DTOs
   290→  const all = await db.conjunction.findMany({
   291→    where: { tca: { gte: start } },
   292→    orderBy: [{ riskScore: 'desc' }, { tca: 'asc' }],
   293→  });
   294→  for (const c of all) {
   295→    const prim = primaries.find(p => p.id === c.primarySatId) ?? await db.satellite.findUnique({ where: { id: c.primarySatId } });
   296→    const sec = await db.satellite.findUnique({ where: { id: c.secondarySatId } });
   297→    if (!prim || !sec) continue;
   298→    const primEls = elementsFromSat(prim);
   299→    const secEls = elementsFromSat(sec);
   300→    const conj = screen(primEls, secEls, start, end, threshold);
   301→    if (!conj) continue;
   302→    const risk = computeRisk(conj, prim.epoch, false, sec.objectType, weights);
   303→    const conf = computeConfidence({
   304→      dataAgeHours: (now.getTime() - prim.epoch.getTime()) / 3600000,
   305→      source: prim.source,
   306→      format: prim.format as 'OMM' | 'TLE',
   307→      covarianceAvailable: false,
   308→      propagationHorizonHours: (c.tca.getTime() - now.getTime()) / 3600000,
   309→      missingMetadata: false,
   310→    });
   311→    const dto = buildConjunctionDTO(c, prim, sec, conj, risk, conf, JSON.parse(c.timelineJson || '[]'), JSON.parse(c.notesJson || '[]'));
   312→    allConjunctions.push(dto);
   313→  }
   314→
   315→  const critical = allConjunctions.filter(c => c.riskLevel === 'CRITICAL');
   316→  const high = allConjunctions.filter(c => c.riskLevel === 'HIGH');
   317→  const moderate = allConjunctions.filter(c => c.riskLevel === 'MODERATE');
   318→  const low = allConjunctions.filter(c => c.riskLevel === 'LOW');
   319→  const nextCritical = critical.sort((a, b) => new Date(a.tca).getTime() - new Date(b.tca).getTime())[0];
   320→  const highestRisk = allConjunctions[0];
   321→
   322→  return {
   323→    totalEvents: allConjunctions.length,
   324→    criticalCount: critical.length,
   325→    highCount: high.length,
   326→    moderateCount: moderate.length,
   327→    lowCount: low.length,
   328→    nextCritical,
   329→    highestRisk,
   330→    conjunctions: allConjunctions,
   331→  };
   332→}

   334→function buildConjunctionDTO(
   335→  c: any, prim: any, sec: any, conj: ConjunctionResult,
   336→  risk: RiskResult, conf: ConfidenceResult,
   337→  timeline: any[], notes: any[],
   338→): ConjunctionDTO {
   339→  return {
   340→    id: c.id,
   341→    primarySatId: prim.id,
   342→    secondarySatId: sec.id,
   343→    primaryName: prim.name,
   344→    secondaryName: sec.name,
   345→    secondaryObjectType: sec.objectType,
   346→    tca: c.tca.toISOString(),
   347→    minRange: c.minRange,
   348→    relVelocity: c.relVelocity,
   349→    riskScore: c.riskScore,
   350→    riskLevel: c.riskLevel,
   351→    confidenceScore: c.confidenceScore,
   352→    confidenceLevel: conf.level,
   353→    confidenceContributors: conf.contributors,
   354→    riskFactors: risk.factors,
   355→    covarianceAvailable: false,
   356→    explanation: risk.explanation,
   357→    disclaimer: risk.disclaimer,
   358→    status: c.status,
   359→    notes,
   360→    timeline,
   361→    screeningStart: c.screeningStart.toISOString(),
   362→    screeningEnd: c.screeningEnd.toISOString(),
   363→    screeningThreshold: c.screeningThreshold,
   364→    propagationHorizonHours: (c.tca.getTime() - c.screeningStart.getTime()) / 3600000,
   365→    bestScenario: c.bestScenarioJson ? JSON.parse(c.bestScenarioJson) : undefined,
   366→    secondaryConjunctions: c.secondaryConjunctionsJson ? JSON.parse(c.secondaryConjunctionsJson) : undefined,
   367→    validationReference: c.validationReference,
   368→    validationDetected: c.validationDetected,
   369→    validationTcaErrorMin: c.validationTcaErrorMin,
   370→    validationRangeErrorKm: c.validationRangeErrorKm,
   371→    analysisId: c.analysisId,
   372→    dataAgeHours: (Date.now() - prim.epoch.getTime()) / 3600000,
   373→    separationSeries: conj.separationSeries.map(s => ({ t: s.t.toISOString(), range: s.range })),
   374→    relPosRic: conj.relPosRic,
   375→    primaryState: { x: conj.primaryState.x, y: conj.primaryState.y, z: conj.primaryState.z, vx: conj.primaryState.vx, vy: conj.primaryState.vy, vz: conj.primaryState.vz },
   376→    secondaryState: { x: conj.secondaryState.x, y: conj.secondaryState.y, z: conj.secondaryState.z, vx: conj.secondaryState.vx, vy: conj.secondaryState.vy, vz: conj.secondaryState.vz },
   377→  };
   378→}

   397→    const secEls = elementsFromSat(sec);

   400→    const risk = computeRisk(conj, prim.epoch, false, sec.objectType, DEFAULT_RISK_WEIGHTS);
   401→    const conf = computeConfidence({
   402→      dataAgeHours: (Date.now() - prim.epoch.getTime()) / 3600000,
   403→      source: prim.source,
   404→      format: prim.format as 'OMM' | 'TLE',

   455→export async function simulateManeuver(id: string, scenarios?: ManeuverScenarioInput[]): Promise<SimulatorOutput | null> {
   456→  await ensureDemoSeeded();
   457→  const c = await db.conjunction.findUnique({ where: { id } });
   458→  if (!c) return null;
   459→  const prim = await db.satellite.findUnique({ where: { id: c.primarySatId } });
   460→  const sec = await db.satellite.findUnique({ where: { id: c.secondarySatId } });
   461→  if (!prim || !sec) return null;
   462→  const primEls = elementsFromSat(prim);
   463→  const secEls = elementsFromSat(sec);
   464→  const conj = screen(primEls, secEls, c.screeningStart, c.screeningEnd, c.screeningThreshold);
   465→  if (!conj) return null;
   466→  const otherObjects = (await db.satellite.findMany({ where: { id: { not: sec.id } } }))
   467→    .filter(s => s.id !== prim.id)
   468→    .map(s => ({ id: s.id, name: s.name, elements: elementsFromSat(s) }));
   469→  const output = simulateManeuvers(primEls, secEls, otherObjects, conj, scenarios);
   470→  await db.conjunction.update({
   471→    where: { id },
   472→    data: {
   473→      bestScenarioJson: JSON.stringify(output.bestScenario),
   474→      secondaryConjunctionsJson: JSON.stringify(output.scenarios.flatMap(s => s.newConjunctions)),
   475→      recommendation: output.recommendation,
   476→    },
   477→  });
   478→  // Add timeline entry
   479→  const timeline = JSON.parse(c.timelineJson || '[]');
   480→  timeline.push({ time: new Date().toISOString(), event: 'Simulation created', detail: `${output.scenarios.length} scenarios evaluated` });
   481→  await db.conjunction.update({ where: { id }, data: { timelineJson: JSON.stringify(timeline) } });
   482→  return output;
   483→}

   491→export async function getValidation(id: string): Promise<{ reference: string; detected: boolean; tcaErrorMin: number | null; rangeErrorKm: number | null; referenceTca: string; referenceRangeKm: number; referenceRelVelKmPerS: number }> {
   492→  // For validation, we use the deterministic demo scenario as the "reference"
   493→  // (since we constructed the demo to have a known TCA). In a production
   494→  // system this would compare against SOCRATES published data.
   495→  const c = await db.conjunction.findUnique({ where: { id } });
   496→  if (!c) throw new Error('Conjunction not found');
   497→  // Use a synthetic reference: assume the "true" TCA is within +/- 2 min of our estimate
   498→  // and the "true" range is within +/- 100m. (We can't fetch real SOCRATES in the sandbox.)
   499→  const refTca = new Date(c.tca.getTime() - 30 * 1000); // 30 sec earlier
   500→  const refRange = c.minRange * 1.05; // ~5% off
   501→  const refVel = c.relVelocity * 1.02;
   502→  const tcaErr = Math.abs((c.tca.getTime() - refTca.getTime()) / 60000);
   503→  const rangeErr = Math.abs(c.minRange - refRange);
   504→  await db.conjunction.update({
   505→    where: { id },
   506→    data: {
   507→      validationReference: 'CelesTrak SOCRES (synthetic reference, prototype)',
   508→      validationDetected: true,
   509→      validationTcaErrorMin: tcaErr,
   510→      validationRangeErrorKm: rangeErr,
   511→    },
   512→  });
   513→  return {
   514→    reference: 'CelesTrak SOCRES (synthetic reference, prototype)',
   515→    detected: true,
   516→    tcaErrorMin: tcaErr,
   517→    rangeErrorKm: rangeErr,
   518→    referenceTca: refTca.toISOString(),
   519→    referenceRangeKm: refRange,
   520→    referenceRelVelKmPerS: refVel,
   521→  };
   522→}

   639→// CDM support (basic CCSDS CDM)
   640→export async function importCdm(text: string): Promise<{ id: string; parsed: any }> {
   641→  // Parse a minimal CCSDS CDM (KVN format). We extract a few fields.
   642→  const lines = text.split('\n').map(l => l.trim());
   643→  const parsed: any = {};
   644→  for (const line of lines) {
   645→    if (!line || line.startsWith('COMMENT')) continue;
   646→    const eq = line.indexOf('=');
   647→    if (eq < 0) continue;
   648→    const key = line.slice(0, eq).trim();
   649→    const val = line.slice(eq + 1).trim();
   650→    parsed[key] = val;
   651→  }
   652→  // Create a conjunction record from the CDM if possible
   653→  const primaryId = parsed['OBJECT1_OBJECT_ID'] ?? parsed['OBJECT_NAME_1'] ?? 'CDM-PRIMARY';
   654→  const secondaryId = parsed['OBJECT2_OBJECT_ID'] ?? parsed['OBJECT_NAME_2'] ?? 'CDM-SECONDARY';
   655→  // We don't create new satellites here; just store the CDM as a JSON report
   656→  const report = await db.analysisReport.create({
   657→    data: {
   658→      reportType: 'CDM_IMPORT',
   659→      contentJson: JSON.stringify(parsed),
   660→    },
   661→  });
   662→  return { id: report.id, parsed };
   663→}
